```
def is_valid_password(s: str):
    """
    Check if the string s contains any leading zeros.
    
    :param s: str
    :return: bool
    """
    from collections import deque
    counter = Counter(s)
    for char in s:
        if char in counter:
            counter[char] += 1
    return counter == 0
```

assert is_valid_password("pass lemon") == False
assert is_valid_password("Password") == True
assert is_valid_password("123456") == False

```python
def can_segment_palindrome(s):
    return s == s[::-1]
```
