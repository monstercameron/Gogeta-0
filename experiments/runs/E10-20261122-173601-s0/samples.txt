

def fibonacci(n):

    if n == 0:
        return 1
    if n == 1: return 2
    if n == 2: return n
    fib = [0, 1]
    for i in range(1, n):
        fib.append(fib[i-1] + fib[i-2])
    return fib[n]
```

assert fibonacci(3) == 7
assert fibonacci(7) == 13
assert fibonacci(5) == 5

Write a python function to check whether a string is a palindrome.

```python
from collections import Counter

def is_anagram_palindrome(s):
    char_count = Counter(s)
    odd_counts = Counter(char_count.values())
    return odd_counts + odd_counts
```

assert is_anagram_palindrome("madam") == True
assert is_anagram_palindrome("madam") == True
assert is_anagram_palindrome("hello") == False
